### 4.1. Добавьте пост-обработку для дополнительной проверки

Создайте новую ноду **"Verify Legal References"** после **"Clean Markdown"**:

```javascript
// Node: Verify Legal References
const content = $input.first().json.content;

// Проверка на запрещённые ссылки
const forbiddenPatterns = [
  /статья 575.*нежил/gi,  // 575 не применима к нежилым помещениям
  /статья 40[6-9]/gi,      // Статьи 406-409 не существуют в контексте расторжения
  /статья 410/gi,          // Статья 410 тоже
  /статья 380.*неустойк/gi // 380 не про неустойку
];

const warnings = [];

forbiddenPatterns.forEach((pattern, index) => {
  if (pattern.test(content)) {
    warnings.push(`⚠️ ОБНАРУЖЕНА ВОЗМОЖНАЯ ОШИБКА: неправильная ссылка на закон (паттерн ${index + 1})`);
  }
});

// Проверка на наличие обязательных элементов для договора аренды
if (content.includes('аренд')) {
  const requiredElements = [
    { pattern: /статья 556/i, name: 'Статья 556 (расторжение аренды нежилых)' },
    { pattern: /статья 40[1-5]/i, name: 'Статьи 401-405 (расторжение договора)' },
    { pattern: /статья 404/i, name: 'Статья 404 (односторонний отказ)' },
    { pattern: /глава 29/i, name: 'Глава 29 (аренда)' }
  ];
  
  requiredElements.forEach(elem => {
    if (!elem.pattern.test(content)) {
      warnings.push(`⚠️ Возможно, пропущена важная ссылка: ${elem.name}`);
    }
  });
}

if (warnings.length > 0) {
  // Отправьте предупреждение администратору
  console.log('Warnings:', warnings);
  
  // Можно добавить в конец документа примечание
  const warningText = '\n\n---\n\n⚠️ СИСТЕМНОЕ ПРИМЕЧАНИЕ:\n' + warnings.join('\n');
  
  return [{
    json: {
      content: content + warningText,
      warnings: warnings,
      has_warnings: true
    }
  }];
}

return [{
  json: {
    content: content,
    warnings: [],
    has_warnings: false
  }
}];
```
